```python
#!/usr/bin/env python3
"""
Real-time arbitrage opportunity detection using Pathway
Processes Solana DEX data streams with <100ms latency
"""

import pathway as pw
import pathway.stdlib.ml.classifiers as ml
from pathway.stdlib.indexing import KNNIndex
from datetime import datetime, timedelta
import json
import asyncio
import aiohttp
from typing import Dict, Any, Optional
import numpy as np

class SolanaArbitragePathway:
    """Real-time arbitrage detection engine using Pathway"""
    
    def __init__(self):
        # Initialize Pathway compute engine
        self.engine = pw.Engine()
        
        # Configuration
        self.MIN_PROFIT_PERCENT = 0.3
        self.MAX_SLIPPAGE = 0.5
        self.CONFIDENCE_THRESHOLD = 0.7
        
    def create_pipeline(self):
        """Create the main Pathway processing pipeline"""
        
        # === DATA INGESTION LAYER ===
        
        # 1. WebSocket stream from Solana
        solana_stream = pw.io.http.rest_connector(
            url="wss://api.mainnet-beta.solana.com",
            streaming_mode=pw.StreamingMode.REALTIME,
            format="json",
            poll_interval_ms=100  # Ultra-low latency
        )
        
        # 2. Raydium price feed
        raydium_prices = pw.io.kafka.read(
            rdkafka_settings={
                "bootstrap.servers": "localhost:9092",
                "group.id": "pathway-arbitrage",
            },
            topic="raydium-prices",
            format="json",
            autocommit_duration_ms=100
        )
        
        # 3. Orca price feed  
        orca_prices = pw.io.kafka.read(
            rdkafka_settings={
                "bootstrap.servers": "localhost:9092",
                "group.id": "pathway-arbitrage",
            },
            topic="orca-prices",
            format="json",
            autocommit_duration_ms=100
        )
        
        # 4. Your bot's trade history (for learning)
        trade_history = pw.io.csv.read(
            "./trades.csv",
            schema=TradeSchema,
            mode="streaming"  # Continuously monitor for new trades
        )
        
        # === TRANSFORMATION & ENRICHMENT LAYER ===
        
        # Parse and normalize price data
        raydium_parsed = raydium_prices.select(
            timestamp=pw.this.timestamp,
            pool_id=pw.this.pool_id,
            token_a=pw.this.token_a,
            token_b=pw.this.token_b,
            price=pw.this.price.as_float(),
            volume=pw.this.volume.as_float(),
            liquidity=pw.this.liquidity.as_float(),
            source=pw.constant("raydium")
        )
        
        orca_parsed = orca_prices.select(
            timestamp=pw.this.timestamp,
            pool_id=pw.this.pool_id,
            token_a=pw.this.token_a,
            token_b=pw.this.token_b,
            price=pw.this.price.as_float(),
            volume=pw.this.volume.as_float(),
            liquidity=pw.this.liquidity.as_float(),
            source=pw.constant("orca")
        )
        
        # === CORE ARBITRAGE DETECTION ===
        
        # Join prices from different DEXs
        price_comparison = raydium_parsed.join(
            orca_parsed,
            pw.left.token_a == pw.right.token_a,
            pw.left.token_b == pw.right.token_b,
            how=pw.JoinMode.INNER
        ).select(
            timestamp=pw.this.left.timestamp,
            token_pair=pw.this.left.token_a + "/" + pw.this.left.token_b,
            raydium_price=pw.this.left.price,
            orca_price=pw.this.right.price,
            raydium_liquidity=pw.this.left.liquidity,
            orca_liquidity=pw.this.right.liquidity,
            price_diff=pw.abs(pw.this.left.price - pw.this.right.price),
            price_diff_percent=(
                pw.abs(pw.this.left.price - pw.this.right.price) / 
                pw.this.left.price * 100
            )
        )
        
        # Calculate arbitrage opportunities
        opportunities = price_comparison.filter(
            pw.this.price_diff_percent > self.MIN_PROFIT_PERCENT
        ).select(
            **pw.this,
            buy_dex=pw.if_else(
                pw.this.raydium_price < pw.this.orca_price,
                "raydium",
                "orca"
            ),
            sell_dex=pw.if_else(
                pw.this.raydium_price < pw.this.orca_price,
                "orca",
                "raydium"
            ),
            estimated_profit=self.calculate_profit,
            confidence_score=self.calculate_confidence,
            risk_score=self.calculate_risk
        )
        
        # === MACHINE LEARNING ENHANCEMENT ===
        
        # Learn from historical trades
        ml_features = trade_history.select(
            price_diff_percent=pw.this.price_diff_percent,
            liquidity_ratio=pw.this.liquidity_ratio,
            volume_24h=pw.this.volume_24h,
            success=pw.this.profit_usd > 0
        )
        
        # Train classifier for opportunity quality
        classifier = ml.LogisticRegression()
        classifier.fit(
            ml_features[["price_diff_percent", "liquidity_ratio", "volume_24h"]],
            ml_features["success"]
        )
        
        # Score new opportunities
        opportunities_scored = opportunities.select(
            **pw.this,
            ml_score=classifier.predict_proba(
                pw.this[["price_diff_percent", "raydium_liquidity", "orca_liquidity"]]
            )
        )
        
        # === RISK MANAGEMENT LAYER ===
        
        # Add temporal windows for rate limiting
        windowed_opportunities = opportunities_scored.windowby(
            pw.this.timestamp,
            window=pw.temporal.sliding(
                duration=timedelta(minutes=5),
                step=timedelta(seconds=30)
            ),
            behavior=pw.temporal.exactly_once()
        ).reduce(
            token_pair=pw.this.token_pair,
            opportunities_count=pw.count(),
            avg_profit=pw.avg(pw.this.estimated_profit),
            max_profit=pw.max(pw.this.estimated_profit),
            avg_confidence=pw.avg(pw.this.confidence_score)
        )
        
        # Filter by confidence and risk
        final_opportunities = windowed_opportunities.filter(
            (pw.this.avg_confidence > self.CONFIDENCE_THRESHOLD) &
            (pw.this.opportunities_count < 10)  # Avoid oversaturation
        )
        
        # === OUTPUT LAYER ===
        
        # 1. Send to execution bot
        pw.io.http.write(
            final_opportunities,
            url="http://localhost:8081/execute",
            format="json",
            request_method="POST"
        )
        
        # 2. Store in database for analysis
        pw.io.postgres.write(
            final_opportunities,
            connection_string="postgresql://user:pass@localhost/arbitrage",
            table_name="opportunities",
            on_conflict="update"
        )
        
        # 3. Send alerts for high-value opportunities
        high_value = final_opportunities.filter(
            pw.this.max_profit > 100  # $100+ profit
        )
        
        pw.io.slack.write(
            high_value,
            webhook_url="YOUR_SLACK_WEBHOOK",
            message_template="🚨 High-value arbitrage: {token_pair} - ${max_profit:.2f} potential!"
        )
        
        return final_opportunities
    
    # === HELPER FUNCTIONS ===
    
    @pw.udf
    def calculate_profit(
        self,
        buy_price: float,
        sell_price: float,
        liquidity_buy: float,
        liquidity_sell: float
    ) -> float:
        """Calculate expected profit considering fees and slippage"""
        
        # Base profit
        spread = sell_price - buy_price
        
        # Trading fees (0.25% each side)
        fees = (buy_price * 0.0025) + (sell_price * 0.0025)
        
        # Slippage estimation based on liquidity
        avg_liquidity = (liquidity_buy + liquidity_sell) / 2
        slippage_factor = min(1.0, avg_liquidity / 1_000_000)  # Normalize to $1M
        estimated_slippage = spread * (1 - slippage_factor) * 0.1
        
        # Gas costs (approximately 0.00025 SOL = $0.04)
        gas_cost = 0.04
        
        # Net profit
        net_profit = spread - fees - estimated_slippage - gas_cost
        
        return max(0, net_profit)
    
    @pw.udf
    def calculate_confidence(
        self,
        price_diff_percent: float,
        liquidity_buy: float,
        liquidity_sell: float,
        volume_24h: Optional[float] = None
    ) -> float:
        """Calculate confidence score for the opportunity"""
        
        scores = []
        
        # Price difference score (higher is better)
        price_score = min(1.0, price_diff_percent / 2.0)
        scores.append(price_score)
        
        # Liquidity score (both sides should have good liquidity)
        min_liquidity = min(liquidity_buy, liquidity_sell)
        liquidity_score = min(1.0, min_liquidity / 100_000)  # $100k baseline
        scores.append(liquidity_score)
        
        # Volume score if available
        if volume_24h:
            volume_score = min(1.0, volume_24h / 1_000_000)  # $1M baseline
            scores.append(volume_score)
        
        # Return weighted average
        return sum(scores) / len(scores)
    
    @pw.udf
    def calculate_risk(
        self,
        liquidity: float,
        price_volatility: float,
        recent_failures: int = 0
    ) -> float:
        """Calculate risk score (0=low risk, 1=high risk)"""
        
        risk_factors = []
        
        # Low liquidity = high risk
        liquidity_risk = max(0, 1 - (liquidity / 500_000))
        risk_factors.append(liquidity_risk)
        
        # High volatility = high risk
        volatility_risk = min(1.0, price_volatility / 10)
        risk_factors.append(volatility_risk)
        
        # Recent failures increase risk
        failure_risk = min(1.0, recent_failures / 5)
        risk_factors.append(failure_risk)
        
        return sum(risk_factors) / len(risk_factors)
    
    def run(self):
        """Start the Pathway processing engine"""
        pipeline = self.create_pipeline()
        pw.run()

class PathwayBotConnector:
    """Bridge between Pathway and your Rust arbitrage bot"""
    
    def __init__(self, bot_endpoint: str = "http://localhost:8081"):
        self.bot_endpoint = bot_endpoint
        
    async def execute_opportunity(self, opportunity: Dict[str, Any]) -> Dict[str, Any]:
        """Send opportunity to Rust bot for execution"""
        
        payload = {
            "action": "execute_arbitrage",
            "params": {
                "buy_dex": opportunity["buy_dex"],
                "sell_dex": opportunity["sell_dex"],
                "token_pair": opportunity["token_pair"],
                "amount_sol": self.calculate_position_size(opportunity),
                "expected_profit": opportunity["estimated_profit"],
                "confidence": opportunity["confidence_score"],
                "max_slippage": 0.5
            }
        }
        
        async with aiohttp.ClientSession() as session:
            async with session.post(
                f"{self.bot_endpoint}/execute",
                json=payload,
                timeout=aiohttp.ClientTimeout(total=30)
            ) as response:
                result = await response.json()
                return result
    
    def calculate_position_size(self, opportunity: Dict[str, Any]) -> float:
        """Dynamic position sizing based on confidence and profit"""
        
        base_position = 10.0  # 10 SOL base
        
        # Scale by confidence
        confidence_multiplier = opportunity["confidence_score"]
        
        # Scale by expected profit
        profit_multiplier = min(2.0, opportunity["estimated_profit"] / 50)
        
        # Final position
        position = base_position * confidence_multiplier * profit_multiplier
        
        # Cap at maximum
        return min(position, 100.0)  # Max 100 SOL
```

### **4. Rust Integration Endpoint (`src/pathway_endpoint.rs`)**

```rust
//! HTTP endpoint for receiving opportunities from Pathway

use actix_web::{web, App, HttpResponse, HttpServer, Result};
use serde::{Deserialize, Serialize};
use log::{info, warn};
use std::sync::Arc;
use tokio::sync::Mutex;

#[derive(Debug, Deserialize)]
pub struct ExecuteRequest {
    pub action: String,
    pub params: OpportunityParams,
}

#[derive(Debug, Deserialize)]
pub struct OpportunityParams {
    pub buy_dex: String,
    pub sell_dex: String,
    pub token_pair: String,
    pub amount_sol: f64,
    pub expected_profit: f64,
    pub confidence: f64,
    pub max_slippage: f64,
}

#[derive(Debug, Serialize)]
pub struct ExecuteResponse {
    pub success: bool,
    pub signature: Option<String>,
    pub error: Option<String>,
}

pub struct PathwayEndpoint {
    executor: Arc<Mutex<crate::executor::TransactionExecutor>>,
}

impl PathwayEndpoint {
    pub fn new(executor: Arc<Mutex<crate::executor::TransactionExecutor>>) -> Self {
        Self { executor }
    }
    
    pub async fn execute_opportunity(
        &self,
        req: web::Json<ExecuteRequest>,
    ) -> Result<HttpResponse> {
        info!("📨 Received opportunity from Pathway: {:?}", req.params);
        
        // Validate opportunity
        if req.params.confidence < 0.5 {
            warn!("⚠️ Confidence too low: {}", req.params.confidence);
            return Ok(HttpResponse::Ok().json(ExecuteResponse {
                success: false,
                signature: None,
                error: Some("Confidence too low".to_string()),
            }));
        }
        
        // Create arbitrage opportunity
        let opportunity = crate::calculator::ArbitrageOpportunity {
            buy_dex: req.params.buy_dex.clone(),
            sell_dex: req.params.sell_dex.clone(),
            buy_price: 0.0,  // Will be fetched in executor
            sell_price: 0.0,
            amount_sol: req.params.amount_sol,
            expected_profit_usd: req.params.expected_profit,
            profit_after_fees_usd: req.params.expected_profit * 0.9,
            profit_percentage: 0.0,
            estimated_gas_sol: 0.00025,
            price_impact: 0.01,
            confidence_score: req.params.confidence,
        };
        
        // Execute trade
        let executor = self.executor.lock().await;
        match executor.execute_arbitrage(&opportunity).await {
            Ok(signature) => {
                info!("✅ Trade executed: {}", signature);
                Ok(HttpResponse::Ok().json(ExecuteResponse {
                    success: true,
                    signature: Some(signature.to_string()),
                    error: None,
                }))
            }
            Err(e) => {
                warn!("❌ Trade failed: {}", e);
                Ok(HttpResponse::Ok().json(ExecuteResponse {
                    success: false,
                    signature: None,
                    error: Some(e.to_string()),
                }))
            }
        }
    }
    
    pub async fn health_check(&self) -> Result<HttpResponse> {
        Ok(HttpResponse::Ok().json(serde_json::json!({
            "status": "healthy",
            "timestamp": chrono::Utc::now().to_rfc3339(),
        })))
    }
}

pub async fn start_pathway_server(
    executor: Arc<Mutex<crate::executor::TransactionExecutor>>,
) -> std::io::Result<()> {
    let endpoint = Arc::new(PathwayEndpoint::new(executor));
    
    HttpServer::new(move || {
        let endpoint = endpoint.clone();
        App::new()
            .route("/execute", web::post().to(move |req| {
                let endpoint = endpoint.clone();
                async move { endpoint.execute_opportunity(req).await }
            }))
            .route("/health", web::get().to(move |_| {
                let endpoint = endpoint.clone();
                async move { endpoint.health_check().await }
            }))
    })
    .bind("127.0.0.1:8081")?
    .run()
    .await
}
```

## 🚀 Uruchomienie Całego Systemu

```bash
# 1. Start Kafka
docker-compose -f docker-compose.kafka.yml up -d

# 2. Start Pathway pipeline
python3 pathway_pipeline.py --mode both &

# 3. Start data producers
python3 producers/solana_producers.py &

# 4. Start your bot with Pathway endpoint
cd /home/marcin/windsurf/Projects/SolanaArbitrageBot
cargo run --release --features pathway-integration

# 5. Monitor in Grafana
open http://localhost:3000
```

## 🎯 Kluczowe Zalety Pathway

1. **Ultra-niska latencja** - <100ms od wykrycia do wykonania
2. **Incremental computation** - Przelicza tylko to co się zmieniło
3. **Time-travel debugging** - Możesz cofnąć się w czasie
4. **ML integration** - Uczy się z historycznych tradów
5. **Automatic backpressure** - Nie przeciąży systemu
